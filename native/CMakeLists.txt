cmake_minimum_required(VERSION 3.24)
project(multimodal_capture_native VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(capture_core src/adapter.cpp src/engine.cpp)
target_include_directories(capture_core PUBLIC include vendor)
if(MSVC)
  target_compile_options(capture_core PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(capture_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(syndaq_engine src/main.cpp)
target_link_libraries(syndaq_engine PRIVATE capture_core)
if(WIN32)
  target_link_libraries(syndaq_engine PRIVATE ws2_32 crypt32 shell32)
  target_compile_definitions(syndaq_engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(syndaq_engine PRIVATE /W4 /utf-8 /EHsc)
endif()

enable_testing()
add_executable(capture_core_test tests/core_test.cpp)
target_link_libraries(capture_core_test PRIVATE capture_core)
if(MSVC)
  target_compile_options(capture_core_test PRIVATE /W4 /utf-8 /EHsc)
endif()
add_test(NAME capture_core_test COMMAND capture_core_test)
